const assert = require('node:assert/strict');
const fs = require('node:fs');
const path = require('node:path');
const ts = require('/opt/nvm/versions/node/v22.16.0/lib/node_modules/typescript/lib/typescript.js');
const root = path.join(__dirname,'..');
for(const f of ['App.js','src/CropPane.jsx','src/cropGesture.js']) {
  const content=fs.readFileSync(path.join(root,f),'utf8');
  const result=ts.transpileModule(content,{
    fileName:f,
    reportDiagnostics:true,
    compilerOptions:{jsx:ts.JsxEmit.React,allowJs:true,target:ts.ScriptTarget.ES2020}
  });
  assert.equal(result.diagnostics.length,0,`${f}: ${result.diagnostics.map(d=>ts.flattenDiagnosticMessageText(d.messageText,' ')).join('; ')}`);
}
const pane=fs.readFileSync(path.join(root,'src/CropPane.jsx'),'utf8');
assert(pane.includes('onResponderGrant={grant}')&&pane.includes('onResponderMove={move}'), 'crop drag surface handles gestures');
assert(pane.includes('collapsable={false}'),'Android retains the interactive crop overlay');
assert(pane.includes('beginCropDrag(')&&pane.includes('updateCropDrag('),'drag lifecycle connected');
assert(pane.includes("from 'expo-image'")&&pane.includes('contentFit="contain"'),'uses the gallery image renderer with original source');
assert(pane.includes('onError={failed}')&&pane.includes('imageError'),'broken image preview reports an error');
assert(pane.includes('!imageLoaded')&&pane.includes('disabled='),'cannot apply invisible image');
const app=fs.readFileSync(path.join(root,'App.js'),'utf8');
assert(app.includes('uri:current.uri,')&&app.includes('inputUri:input.uri'),'crop shows gallery image, with a separate processing copy');
assert(app.includes('ImageManipulator.manipulateAsync('),'use documented preview normalization with a real file');
assert(app.includes('if(!demo && cropPreview)')&&app.includes("uri.startsWith('file://')"),'cleanup only touches editor-owned bytes, never the user image');
assert(app.includes('fallbackUri={cropPreview.inputUri}'),'original has a byte-for-byte local fallback');
assert(app.includes('const op=await replaceWithCrop(current,processed.uri,sourceUri)'),'crop writes via safe SAF transaction');
console.log('PASS v17 editor: JSX syntax, touch handling, gallery preview, guarded apply, owner-safe cleanup, SAF save');
